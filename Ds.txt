1(a). STACK ADT USING ARRAY


CODE:
#include <stdio.h>
#define MAX 5
int stack[MAX], top = -1;
 void push(int value)  {
     if (top == MAX - 1)  {
         printf("Stack Overflow");
         return;
    }
     stack[++top] = value;
     printf("%d pushed", value);
}
 void pop(void)  {
     if (top == -1)  {
         printf("Stack Underflow");
         return;
    }
     printf("Popped: %d", stack[top--]);
}
 void display(void)  {
     int i;
     if (top == -1)  {
         printf("Stack is empty");
         return;
    }
     printf("Stack:");
     for (i = top; i >= 0; i--) printf("%d ", stack[i]);
     printf("\n");
}
 int main(void)  {
     push(10);
     push(20);
     push(30);
     display();
     pop();
     display();
     return 0;
}
OUTPUT:  10 pushed 20 pushed 30 pushed Stack: 30 20 10 Popped: 30 Stack: 20 10

CODE:
#include <stdio.h>
#define MAX 5
int queue[MAX], front = -1, rear = -1;
 void enqueue(int value)  {
     if ((rear + 1) % MAX == front)  {
         printf("Queue Overflow");
         return;
    }
     if (front == -1) front = 0;
     rear = (rear + 1) % MAX;
     queue[rear] = value;
     printf("%d inserted", value);
}
 void dequeue(void)  {
     if (front == -1)  {
         printf("Queue Underflow");
         return;
    }
     printf("Deleted: %d", queue[front]);
     if (front == rear)  {
         front = rear = -1;
    }
     else  {
         front = (front + 1) % MAX;
    }
}
 void display(void)  {
     int i;
     if (front == -1)  {
         printf("Queue is empty");
         return;
    }
     printf("Queue:");
     i = front;
     while (1)  {
         printf("%d ", queue[i]);
         if (i == rear) break;
         i = (i + 1) % MAX;
    }
     printf("\n");
}
 int main(void)  {
     enqueue(10);
     enqueue(20);
     enqueue(30);
     display();

4. POLYNOMIAL MANIPULATION USING LINKED LIST


CODE:
#include <stdio.h> 
#include <stdlib.h>
struct Node  {
     int coeff, exp;
     struct Node *next;
};
 struct Node *insert(struct Node head, int c, int e)  {
     struct Node *n = malloc(sizeof(struct Node));
     struct Node *t;
     if (n == NULL) return head;
     n->coeff = c;
     n->exp = e;
     n->next = NULL;
     if (head == NULL) return n;
     t = head;
     while (t->next != NULL) t = t->next;
     t->next = n;
     return head;
}
 struct Node *add(struct Node a, struct Node b)  {
     struct Node *result = NULL;
     while (a != NULL && b != NULL)  {
         if (a->exp == b->exp)  {
             result = insert(result, a->coeff + b->coeff, a->exp);
             a = a->next;
             b = b->next;
        }
         else if (a->exp > b->exp)  {
             result = insert(result, a->coeff, a->exp);
             a = a->next;
        }
         else  {
             result = insert(result, b->coeff, b->exp);
             b = b->next;
        }
    }
     while (a != NULL)  {
         result = insert(result, a->coeff, a->exp);
         a = a->next;
    }
     while (b != NULL)  {
         result = insert(result, b->coeff, b->exp);
         b = b->next;
    }
     return result;
}
 void display(struct Node *p)  {
     while (p != NULL)  {
         printf("%dx^%d", p->coeff, p->exp);
         p = p->next;
         if (p != NULL) printf(" + ");
    }
     printf("\n");
}
 int main(void)  {
     struct Node *p1 = NULL, *p2 = NULL, *sum;
     p1 = insert(p1, 5, 2);
     p1 = insert(p1, 4, 1);
     p1 = insert(p1, 2, 0);
     p2 = insert(p2, 3, 2\);
     p2 = insert(p2, 2, 1);
     p2 = insert(p2, 1, 0);
     printf("First polynomial: ");
     display(p1);
     printf("Second polynomial: ");
     display(p2);
     sum = add(p1, p2);
     printf("Sum: ");
     display(sum);
     return 0;
}
OUTPUT:  First polynomial: 5x^2 + 4x^1 + 2x^0 Second polynomial: 3x^2 + 2x^1 + 1x^0 Sum: 8x^2 + 6x^1 + 3x^0

     char exp[50];
     int i, a, b, result;
     printf("Enter postfix expression:");
     scanf("%49s", exp);
     for (i = 0; exp[i] != '\0'; i++)  {
         if (isdigit((unsigned char)exp[i]))  {
             push(exp[i] - '0');
        }
         else  {
             b = pop();
             a = pop();
             switch (exp[i])  {
                 case '+': result = a + b;
                 break;
                 case '-': result = a - b;
                 break;
                 case '*': result = a * b;
                 break;
                 case '/': if (b == 0)  {
                     printf("Division by zero\n");
                     return 1;
                }
                 result = a / b;
                 break;
                 default: printf("Invalid operator\n");
                 return 1;
            }
             push(result);
        }
    }

#include <ctype.h>
char stack[50];
 int top = -1;
 int precedence(char op)  {
     if (op == '^') return 3;
     if (op == '*' || op == '/') return 2;
     if (op == '+' || op == '-') return 1;
     return 0;
}
 int main(void)  {
     char infix[50], postfix[50];
     int i, k = 0;
     char ch;
     printf("Enter infix expression:");
     scanf("%49s", infix);
     for (i = 0; infix[i] != '\0'; i++)  {
         ch = infix[i];
         if (isalnum((unsigned char)ch))  {
             postfix[k++] = ch;
        }
         else if (ch == '(')  {
             stack[++top] = ch;
        }
         else if (ch == ')')  {
             while (top != -1 && stack[top] != '(') postfix[k++] = stack[top--];
             if (top != -1) top--;
        }
         else  {
             while (top != -1 && stack[top] != '(' && (precedence(stack[top]) > precedence(ch) || (precedence(stack[top]) == precedence(ch) && ch != '^'))) postfix[k++] = stack[top--];
             stack[++top] = ch;
        }
    }
     while (top != -1) postfix[k++] = stack[top--];
     postfix[k] = '\0';

7. AVL TREE


CODE:
#include <stdio.h> 
#include <stdlib.h>
struct Node  {
     int key, height;
     struct Node *left, *right;
};
 int max(int a, int b)  {
     return a > b ? a : b;
}
 int height(struct Node *n)  {
     return n == NULL ? 0 : n->height;
}
 struct Node *newNode(int key)  {
     struct Node *n = malloc(sizeof(struct Node));
     if (n == NULL) return NULL;
     n->key = key;
     n->left = n->right = NULL;
     n->height = 1;
     return n;
}
 struct Node *rightRotate(struct Node *y)  {
     struct Node *x = y->left, *t2 = x->right;
     x->right = y;
     y->left = t2;
     y->height = 1 + max(height(y->left), height(y->right));
     x->height = 1 + max(height(x->left), height(x->right));
     return x;
}
 struct Node *leftRotate(struct Node *x)  {
     struct Node *y = x->right, *t2 = y->left;
     y->left = x;
     x->right = t2;
     x->height = 1 + max(height(x->left), height(x->right));
     y->height = 1 + max(height(y->left), height(y->right));
     return y;
}
 int balance(struct Node *n)  {
     return n == NULL ? 0 : height(n->left) - height(n->right);
}
 struct Node *insert(struct Node *node, int key)  {
     int b;
     if (node == NULL) return newNode(key);
     if (key < node->key) node->left = insert(node->left, key);
     else if (key > node->key) node->right = insert(node->right, key);
     else return node;
     node->height = 1 + max(height(node->left), height(node->right));
     b = balance(node);
     if (b > 1 && key < node->left->key) return rightRotate(node);
     if (b < -1 && key > node->right->key) return leftRotate(node);
     if (b > 1 && key > node->left->key)  {
         node->left = leftRotate(node->left);
         return rightRotate(node);
    }
     if (b < -1 && key < node->right->key)  {
         node->right = rightRotate(node->right);
         return leftRotate(node);
    }
     return node;
}
 void preorder(struct Node *root)  {
     if (root != NULL)  {
         printf("%d ", root->key);
         preorder(root->left);
         preorder(root->right);
    }
}
 int main(void)  {
     struct Node *root = NULL;
     root = insert(root, 10);
     root = insert(root, 20);
     root = insert(root, 30);
     root = insert(root, 40);
     root = insert(root, 50);
     root = insert(root, 25);
     printf("Preorder traversal: ");
     preorder(root);
     printf("\n");
     return 0;
}

8. HEAP USING PRIORITY QUEUE


CODE:
#include <stdio.h>
#define MAX 20
int heap[MAX], size = 0;
 void insert(int value)  {
     int i, parent;
     if (size == MAX)  {
         printf("Priority queue is full");
         return;
    }
     i = size++;
     while (i > 0)  {
         parent = (i - 1) / 2;
         if (heap[parent] >= value) break;
         heap[i] = heap[parent];
         i = parent;
    }
     heap[i] = value;
}
 int deleteMax(void)  {
     int maxValue, last, i = 0, child;
     if (size == 0)  {
         printf("Priority queue is empty");
         return -1;
    }
     maxValue = heap[0];
     last = heap[--size];
     while (2 * i + 1 < size)  {
         child = 2 * i + 1;
         if (child + 1 < size && heap[child + 1] > heap[child]) child++;
         if (last >= heap[child]) break;
         heap[i] = heap[child];
         i = child;
    }
     if (size > 0) heap[i] = last;
     return maxValue;
}
 void display(void)  {
     int i;
     printf("Heap:");
     for (i = 0; i < size; i++) printf("%d ", heap[i]);
     printf("\n");
}
 int main(void)  {
     insert(30);
     insert(10);
     insert(50);
     insert(20);
     insert(40);
     display();
     printf("Deleted highest priority: %d", deleteMax());
     display();
     return 0;
}
OUTPUT:  Heap: 50 40 30 10 20 Deleted highest priority: 50 Heap: 40 20 30 10

#include <stdio.h>
#define V 5
#define INF 99999
int main(void)  {
     int graph[V][V] =  {
          {
            0, 10, 0, 5, 0
        },  {
            10, 0, 1, 2, 0
        },  {
            0, 1, 0, 0, 4
        },  {
            5, 2, 0, 0, 2
        },  {
            0, 0, 4, 2, 0
        }
    };
     int dist[V], visited[V] =  {
        0
    };
     int i, count, u = -1, min, v, source = 0;
     for (i = 0; i < V; i++) dist[i] = INF;
     dist[source] = 0;
     for (count = 0; count < V - 1; count++)  {
         min = INF;
         u = -1;
         for (i = 0; i < V; i++)  {
             if (!visited[i] && dist[i] < min)  {
                 min = dist[i];
                 u = i;
            }
        }
         if (u == -1) break;
         visited[u] = 1;
         for (v = 0; v < V; v++)  {
             if (!visited[v] && graph[u][v] != 0 && dist[u] != INF && dist[u] + graph[u][v] < dist[v])  {
                 dist[v] = dist[u] + graph[u][v];
            }
        }
    }
     printf("Shortest distances from vertex %d:\n", source);
     for (i = 0; i < V; i++) printf("To %d = %d\n", i, dist[i]);
     return 0;
}
OUTPUT:  Shortest distances from vertex 0: To 0 = 0 To 1 = 7 To 2 = 8 To 3 = 5 To 4 = 7

          {

        },  {
            2, 0, 3, 8, 5
        },  {
            0, 3, 0, 0, 7
        },  {

        },  {
            0, 5, 7, 9, 0
        }
    };
     int selected[V] =  {
        0
    };
     int edges = 0, i, j, min, x = -1, y = -1, total = 0;
     selected[0] = 1;
     printf("Edges in minimum spanning tree:\n");
     while (edges < V - 1)  {
         min = INF;
         x = y = -1;
         for (i = 0; i < V; i++)  {
             if (selected[i])  {
                 for (j = 0; j < V; j++)  {
                     if (!selected[j] && graph[i][j] != 0 && graph[i][j] < min)  {
                         min = graph[i][j];
                         x = i;
                         y = j;
                    }
                }
            }
        }
         if (x == -1)  {
             printf("Graph is not connected\n");
             return 1;
        }
         printf("%d - %d : %d\n", x, y, min);
         total += min;
         selected[y] = 1;
         edges++;
    }
     printf("Total cost: %d\n", total);

void merge(int a[], int left, int mid, int right)  {

    }
     while (i <= mid) temp[k++] = a[i++];
     while (j <= right) temp[k++] = a[j++];
     for (x = 0; x < k; x++) a[left + x] = temp[x];
}
 void mergeSort(int a[], int left, int right)  {
     int mid;
     if (left < right)  {
         mid = left + (right - left) / 2;
         mergeSort(a, left, mid);
         mergeSort(a, mid + 1, right);
         merge(a, left, mid, right);
    }
}
 int main(void)  {
     int a[] =  {
        38, 27, 43, 3, 9, 82, 10
    };
     int n = 7, i;
     mergeSort(a, 0, n - 1);
     printf("Sorted array: ");
     for (i = 0; i < n; i++) printf("%d ", a[i]);
     printf("\n");
     return 0;
}
OUTPUT:  Sorted array: 3 9 10 27 38 43 82

14(a). HASHING USING LINEAR PROBING


CODE:
#include <stdio.h>
#define SIZE 10 int main(void)  {
     int table[SIZE], keys[] =  {
        23, 43, 13, 27
    };
     int n = 4, i, index, j, placed;
     for (i = 0; i < SIZE; i++) table[i] = -1;
     for (i = 0; i < n; i++)  {
         index = keys[i] % SIZE;
         placed = 0;
         for (j = 0; j < SIZE; j++)  {
             int pos = (index + j) % SIZE;
             if (table[pos] == -1)  {
                 table[pos] = keys[i];
                 placed = 1;
                 break;
            }
        }
         if (!placed) printf("Hash table is full; %d not inserted\n", keys[i]);
    }
     printf("Index Value\n");
     for (i = 0; i < SIZE; i++) printf("%5d %5d\n", i, table[i]);
     return 0;
}
OUTPUT:  Index Value 0 -1 1 -1 2 -1 3 23 4 43 5 13 6 -1 7 27 8 -1 9 -1
